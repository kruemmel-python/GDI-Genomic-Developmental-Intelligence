cmake_minimum_required(VERSION 3.20)
project(R15MultiTarget VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
configure_file(config.hpp.in generated/config.hpp @ONLY)
add_library(r15_math STATIC src/math.cpp)
target_include_directories(r15_math PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(r15_math PUBLIC R15_CONTRACT=1)
add_executable(r15_app src/main.cpp)
target_link_libraries(r15_app PRIVATE r15_math)
enable_testing()
add_executable(r15_math_test tests/math_test.cpp)
target_link_libraries(r15_math_test PRIVATE r15_math)
add_test(NAME r15_math_positive COMMAND r15_math_test 2 3 5)
add_test(NAME r15_math_negative COMMAND r15_math_test -2 1 -1)
